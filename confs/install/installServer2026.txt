# appending .domain is the MUA's job.
append_dot_mydomain = no

# Uncomment the next line to generate "delayed mail" warnings
#delay_warning_time = 4h

alias_database = hash:/etc/aliases
# Any aliases that are supported by this system
virtual_alias_maps = pgsql:/etc/postfix/pgsql/virtual_alias_maps.cf
# Look up the mailbox location based on the email address received, result must end with a '/' to be a maildir
virtual_mailbox_maps = pgsql:/etc/postfix/pgsql/virtual_mailbox_maps.cf
mydestination = $myhostname, localhost.$mydomain, localhost, sciz.brion.fr
relayhost =
mynetworks = all
inet_interfaces = all
recipient_delimiter = +
##
# Filter out anything that is not mountyhall related
##
header_checks = regexp:/etc/postfix/header_checks

compatibility_level = 3.6

# SCIZ Postfix configuration
# Tells postfix where to find aliases
###
user            = sciz
password        = xxx
dbname          = sciz
hosts           = sql_sciz
query           = SELECT mail FROM sciz.user WHERE mail='%u@%d' or mail='%u@sciz.fr' LIMIT 1;
#############################
/etc/postfix/pgsql/virtual_mailbox_maps.cf
#############################
###
# SCIZ Postfix configuration
# Tells postfix how to find a user Maildir/
###
user            = sciz
password        = xxx
dbname          = sciz
hosts           = sql_sciz
query           = SELECT mail || '/' FROM sciz.user WHERE mail='%u@%d' or mail='%u@sciz.fr'  LIMIT 1;
#############################
/etc/postfix/header_checks
#############################
if /^Subject:/
/^Subject:.*(\[MountyHall\]|W01vdW50eUhhbGxd)/ PASS because MountyHall
/^Subject:.*(Gmail|GMAIL|gmail)/ PASS because Gmail code
/^Subject:.*(Proton|PROTON|proton)/ PASS because may be proton validation
!/^Subject:.*(\[MountyHall\]|W01vdW50eUhhbGxd)/ DISCARD because not MountyHall
endif
